/**
 * @brief muestra la contraseÃ±a en asteriscos
 * @return retorna la contraseÃ±a como string
 */

string leerContrasena() {
    string clave;
    int tecla;
    while ((tecla = _getch()) != 13) {      // 13 = Enter
        if (tecla == 8) {                   // 8 = Retroceso
            if (!clave.empty()) {
                clave.erase(clave.size() - 1);
                cout << "\b \b";
            }
        } else if (tecla >= 32 && tecla <= 126) {
            clave += (char)tecla;
            cout << '*';
        }
    }
    cout << endl;
    return clave;
}



int main() {
    try {
        inicializarArchivoUsuarios();
        inicializarArchivoHabitaciones();
        inicializarArchivoHuespedes();
        inicializarArchivoServicios();
        inicializarArchivoReservas();
        inicializarArchivoConsumos();
        inicializarArchivoFacturas();
    } catch (const exception& e) {
        cout << "Error al preparar los archivos: " << e.what() << endl;
        return 1;
    }

    cout << "--- SISTEMA HOTEL CONSOLE ---" << endl;
    int intentos = 0;
    while (intentos < 3) {
        string nombre;
        cout << "Usuario: ";
        if (!getline(cin, nombre)) return 0;
        cout << "Contrasena: ";
        string clave = leerContrasena();

        Usuario* usuario = 0;
        try {
            usuario = iniciarSesion(nombre, clave);
        } catch (const exception& e) {
            cout << e.what() << "\n" << endl;
            intentos++;
        }
        if (usuario != 0) {
            menuPrincipal(*usuario);
            delete usuario;
            return 0;
        }
    }
    cout << "Demasiados intentos fallidos. Programa terminado." << endl;
    return 1;
}

#include "../header/Alimentacion.h"
#include "../header/utilidades.h"
#include <stdexcept>
#include <sstream>

using namespace std;

Alimentacion::Alimentacion() : Servicio(), tipoComida("") {}

Alimentacion::Alimentacion(string codigo, string nombre, string descripcion,
                           double precio, bool activo, string tipoComida)
    : Servicio(codigo, nombre, descripcion, precio, activo), tipoComida(tipoComida) {
    if (tipoComida.empty() || tipoComida.find_first_of(",\r\n") != string::npos)
        throw invalid_argument("El tipo de comida es obligatorio y no puede contener comas.");
}

string Alimentacion::getTipoComida() const { return tipoComida; }
string Alimentacion::getUnidadCobro() const { return "plato"; }
string Alimentacion::getTipoServicio() const { return "Alimentacion"; }

string Alimentacion::describir() const {
    return "[" + codigo + "] " + nombre + " (" + tipoComida + ") - $" +
           dinero(precio) + " por " + getUnidadCobro();
}

// Costo = precio del plato por la cantidad de platos.
double Alimentacion::calcularCosto(int cantidad) const {
    if (cantidad <= 0) throw invalid_argument("La cantidad debe ser mayor a cero.");
    if (!activo) throw invalid_argument("El servicio de alimentacion esta inactivo.");
    return precio * cantidad;
}

string Alimentacion::serializar() const {
    return Servicio::serializar() + "," + tipoComida;
}

void Alimentacion::deserializar(const string& linea) {
    stringstream ss(linea);
    string tipo, cod, nom, d, precStr, actStr, comida;
    if (!getline(ss, tipo, ',') || !getline(ss, cod, ',') ||
        !getline(ss, nom, ',') || !getline(ss, d, ',') ||
        !getline(ss, precStr, ',') || !getline(ss, actStr, ',') ||
        !getline(ss, comida)) {
        throw invalid_argument("Formato CSV invalido para Alimentacion.");
    }
    if (!comida.empty() && comida.back() == '\r') comida.pop_back();
    if (tipo != "Alimentacion")
        throw invalid_argument("La linea no corresponde a un servicio de Alimentacion.");

    Alimentacion temp(cod, nom, d, stod(precStr), actStr == "1", comida);
    *this = temp;
}

string Fecha::hoy() {
    time_t ahora = time(0);
    tm* f = localtime(&ahora);
    char texto[11];
    strftime(texto, sizeof(texto), "%Y-%m-%d", f);
    return string(texto);

bool Fecha::esBisiesto(int anio) {
    if (anio % 400 == 0) return true;
    if (anio % 100 == 0) return false;
    if (anio % 4 == 0) return true;
    return false;
}

int Fecha::diasDelMes(int mes, int anio) {
    if (mes == 2) {
        if (esBisiesto(anio)) return 29;
        return 28;
    }

    if (mes == 4 || mes == 6 || mes == 9 || mes == 11) {
    return 30;
    }

    return 31;
}

bool Fecha::esValida(string fecha) {
    if (fecha.length() != 10) return false;
    if (fecha[4] != '-' || fecha[7] != '-') return false;

    for (int i = 0; i < 10; i++) {
        if (i != 4 && i != 7) {
            if (fecha[i] < '0' || fecha[i] > '9') return false;
        }
    }

    int anio = stoi(fecha.substr(0, 4));
    int mes = stoi(fecha.substr(5, 2));
    int dia = stoi(fecha.substr(8, 2));

    if (anio < 1) return false;
    if (mes < 1 || mes > 12) return false;
    if (dia < 1 || dia > diasDelMes(mes, anio)) return false;

    return true;
}

int Fecha::contarDias(string fecha) {
    int anio = stoi(fecha.substr(0, 4));
    int mes = stoi(fecha.substr(5, 2));
    int dia = stoi(fecha.substr(8, 2));

    int total = dia;


    for (int a = 1; a < anio; a++) {
        total = total + 365;
        if (esBisiesto(a)) total = total + 1;
    }
    for (int m = 1; m < mes; m++) {
        total = total + diasDelMes(m, anio);
    }

    return total;
}
int Fecha::diasEntre(string fechaEntrada, string fechaSalida) {
    if (!esValida(fechaEntrada) || !esValida(fechaSalida)) {
        throw invalid_argument("La fecha no es valida.");
    }

    int noches = contarDias(fechaSalida) - contarDias(fechaEntrada);

    if (noches <= 0) {
        throw invalid_argument("La salida debe ser posterior a la entrada.");
    }

    return noches;
}

bool Fecha::seSuperponen(string entrada1, string salida1,string entrada2, string salida2) {
    diasEntre(entrada1, salida1);
    diasEntre(entrada2, salida2);

    if (entrada1 < salida2 && entrada2 < salida1) {
        return true;
    }

    return false;
}

static int buscarIndice(const vector<Habitacion>& lista, const string& codigo) {
    for (size_t i = 0; i < lista.size(); i++)
        if (lista[i].getCodigo() == codigo) return (int)i;
    return -1;
}

// Cuenta las reservas de una habitacion. Si soloActivas es true, ignora las canceladas.
static int contarReservasDeHabitacion(const string& codigo, bool soloActivas) {
    vector<Reserva> reservas = cargarReservas();
    int cantidad = 0;
    for (size_t i = 0; i < reservas.size(); i++) {
        if (reservas[i].getCodigoHabitacion() != codigo) continue;
        if (soloActivas && !reservas[i].estaActiva()) continue;
        cantidad++;
    }
    return cantidad;
}

static TipoHabitacion elegirTipo() {
    int t = leerEntero("Tipo (1=Sencilla, 2=Doble, 3=Familiar, 4=Suite): ", 1, 4);
    if (t == 1) return TipoHabitacion::Sencilla;
    if (t == 2) return TipoHabitacion::Doble;
    if (t == 3) return TipoHabitacion::Familiar;
    return TipoHabitacion::Suite;
}

}

static void agregarHabitacionMenu() {

    int capacidad = leerEntero("Capacidad (1-20): ", 1, 20);
    double precio = leerDecimal("Precio por noche: ", 0);

static void actualizarHabitacionMenu() {
    string codigo = leerTextoObligatorio("Codigo de la habitacion: ");
    vector<Habitacion> lista = cargarHabitaciones();
    int pos = buscarIndice(lista, codigo);
    if (pos == -1) throw invalid_argument("No existe esa habitacion.");
    TipoHabitacion tipo = elegirTipo();
    int capacidad = leerEntero("Nueva capacidad (1-20): ", 1, 20);
    double precio = leerDecimal("Nuevo precio por noche: ", 0);

    // Si baja la capacidad, no puede dejar a una reserva activa con mas personas de las que caben.
    vector<Reserva> reservas = cargarReservas();
    for (size_t i = 0; i < reservas.size(); i++) {
        if (reservas[i].getCodigoHabitacion() == codigo && reservas[i].estaActiva() &&
            reservas[i].getPersonas() > capacidad)
            throw invalid_argument("La reserva " + reservas[i].getCodigo() +
                                   " tiene mas personas que la nueva capacidad.");
    }
    lista[pos].actualizarDatos(tipo, capacidad, precio);
    guardarHabitaciones(lista);
    cout << "Habitacion actualizada.\n";
}

static void cambiarEstadoMenu() {
    string codigo = leerTextoObligatorio("Codigo de la habitacion: ");
    vector<Habitacion> lista = cargarHabitaciones();
    int pos = buscarIndice(lista, codigo);
    if (pos == -1) throw invalid_argument("No existe esa habitacion.");
    int e = leerEntero("Estado (1=Disponible, 2=Mantenimiento, 3=Fuera de servicio): ", 1, 3);
    if (e != 1 && contarReservasDeHabitacion(codigo, true) > 0)
        throw invalid_argument("La habitacion tiene reservas activas. Cancelalas antes de bloquearla.");
    if (e == 1) lista[pos].cambiarEstado(EstadoHabitacion::Disponible);
    else if (e == 2) lista[pos].cambiarEstado(EstadoHabitacion::Mantenimiento);
    else lista[pos].cambiarEstado(EstadoHabitacion::FueraDeServicio);
    guardarHabitaciones(lista);
    cout << "Estado actualizado.\n";
}

static void eliminarHabitacionMenu() {
    string codigo = leerTextoObligatorio("Codigo de la habitacion a eliminar: ");
    vector<Habitacion> lista = cargarHabitaciones();
    int pos = buscarIndice(lista, codigo);
    if (pos == -1) throw invalid_argument("No existe esa habitacion.");
    if (contarReservasDeHabitacion(codigo, false) > 0)
        throw invalid_argument("Tiene historial de reservas. Mejor ponla fuera de servicio.");
    if (confirmar("Eliminar la habitacion " + codigo)) {
        lista.erase(lista.begin() + pos);
        guardarHabitaciones(lista);
        cout << "Habitacion eliminada.\n";
    }
}

static void reporteOcupacion() {
    vector<Habitacion> habitaciones = cargarHabitaciones();
    vector<Reserva> reservas = cargarReservas();
    vector<string> lineas;
    lineas.push_back("REPORTE DE OCUPACION");
    lineas.push_back("Fecha del reporte: " + Fecha::hoy());
    lineas.push_back("");
    int ocupadasHoy = 0;
    for (size_t i = 0; i < habitaciones.size(); i++) {
        int reservasActivas = 0;
        int nochesReservadas = 0;
        for (size_t j = 0; j < reservas.size(); j++) {
            if (reservas[j].getCodigoHabitacion() == habitaciones[i].getCodigo() &&
                reservas[j].estaActiva()) {
                reservasActivas++;
                nochesReservadas = nochesReservadas + reservas[j].calcularNoches();
            }
        }
        string estado = estadoActualHabitacion(habitaciones[i], reservas);
        if (estado == "Ocupada") ocupadasHoy++;
        lineas.push_back("Hab " + habitaciones[i].getCodigo() + " | " +
                         Habitacion::nombreTipo(habitaciones[i].getTipo()) +
                         " | Hoy: " + estado +
                         " | Reservas activas: " + to_string(reservasActivas) +
                         " | Noches reservadas: " + to_string(nochesReservadas));
    }
    lineas.push_back("");
    lineas.push_back("Habitaciones ocupadas hoy: " + to_string(ocupadasHoy) +
                     " de " + to_string(habitaciones.size()));
    escribirLineas(rutaArchivo("reporte_ocupacion.txt"), lineas);
}

// Una habitacion esta libre si ninguna reserva activa suya se cruza con las fechas pedidas.
// ignorarCodigo sirve al modificar una reserva (para no chocar consigo misma).
bool habitacionLibre(const string& codigoHabitacion, const string& entrada,
                     const string& salida, const vector<Reserva>& reservas,
                     const string& ignorarCodigo) {
    for (size_t i = 0; i < reservas.size(); i++) {
        const Reserva& r = reservas[i];
        if (!r.estaActiva()) continue;
        if (r.getCodigoHabitacion() != codigoHabitacion) continue;
        if (r.getCodigo() == ignorarCodigo) continue;
        if (Fecha::seSuperponen(entrada, salida, r.getFechaEntrada(), r.getFechaSalida()))
            return false;
    }
    return true;
}

// tipo = "" significa cualquier tipo de habitacion.
vector<Habitacion> buscarHabitacionesLibres(const string& entrada, const string& salida,
                                            int personas, const string& tipo) {
    Fecha::diasEntre(entrada, salida); // valida las fechas
    if (personas <= 0) throw invalid_argument("El numero de personas debe ser positivo.");
    vector<Habitacion> habitaciones = cargarHabitaciones();
    vector<Reserva> reservas = cargarReservas();
    vector<Habitacion> libres;
    for (size_t i = 0; i < habitaciones.size(); i++) {
        const Habitacion& h = habitaciones[i];
        if (!h.estaOperativa()) continue;                 // mantenimiento o fuera de servicio
        if (h.getCapacidad() < personas) continue;        // no cabe la gente
        if (tipo != "" && Habitacion::nombreTipo(h.getTipo()) != tipo) continue;
        if (habitacionLibre(h.getCodigo(), entrada, salida, reservas, ""))
            libres.push_back(h);
    }
    return libres;
}

string estadoActualHabitacion(const Habitacion& h, const vector<Reserva>& reservas) {
    if (!h.estaOperativa()) return Habitacion::nombreEstado(h.getEstado());
    string hoy = Fecha::hoy();
    for (size_t i = 0; i < reservas.size(); i++) {
        const Reserva& r = reservas[i];
        if (r.estaActiva() && r.getCodigoHabitacion() == h.getCodigo() &&
            r.getFechaEntrada() <= hoy && hoy < r.getFechaSalida())
            return "Ocupada";
    }
    return "Disponible";
}

static Servicio* crearServicioDesdeLinea(const string& linea) {
    vector<string> c = dividirCampos(linea);
    if (c.empty()) throw invalid_argument("Linea de servicio vacia.");
    if (c[0] == "Alimentacion") {
        Alimentacion* s = new Alimentacion();
        try { s->deserializar(linea); } catch (...) { delete s; throw; }
        return s;
    }
    if (c[0] == "Lavanderia") {
        Lavanderia* s = new Lavanderia();
        try { s->deserializar(linea); } catch (...) { delete s; throw; }
        return s;
    }
    if (c[0] == "Parqueadero") {
        Parqueadero* s = new Parqueadero();
        try { s->deserializar(linea); } catch (...) { delete s; throw; }
        return s;
    }
    throw invalid_argument("Tipo de servicio desconocido: " + c[0]);
}

void liberarServicios(vector<Servicio*>& lista) {
    for (size_t i = 0; i < lista.size(); i++) delete lista[i];
    lista.clear();
}

static int buscarServicio(const vector<Servicio*>& lista, const string& codigo) {
    for (size_t i = 0; i < lista.size(); i++)
        if (lista[i]->getCodigo() == codigo) return (int)i;
    return -1;
}

vector<Servicio*> cargarServicios() {
    vector<string> lineas = leerLineas(rutaServicios());
    vector<Servicio*> lista;
    try {
        for (size_t i = 0; i < lineas.size(); i++) {
            Servicio* s = crearServicioDesdeLinea(lineas[i]);
            if (buscarServicio(lista, s->getCodigo()) != -1) {
                delete s;
                throw invalid_argument("codigo de servicio repetido en la linea " + to_string(i + 1));
            }
            lista.push_back(s);
        }
    } catch (const exception& e) {
        liberarServicios(lista);
        throw runtime_error(string("servicios.csv: ") + e.what());
    }
    return lista;
}

static Usuario* crearUsuario(const string& nombre, const string& clave,

static void liberarUsuarios(vector<Usuario*>& lista) {
    for (size_t i = 0; i < lista.size(); i++) delete lista[i];
    lista.clear();
}

static int buscarUsuario(const vector<Usuario*>& lista, const string& nombre) {
    for (size_t i = 0; i < lista.size(); i++)
        if (lista[i]->getNombreUsuario() == nombre) return (int)i;
    return -1;
}

static vector<Usuario*> cargarUsuarios() {
    vector<string> lineas = leerLineas(rutaUsuarios());
    vector<Usuario*> lista;
    try {
        for (size_t i = 0; i < lineas.size(); i++) {
            vector<string> c = dividirCampos(lineas[i]);
            if (c.size() < 3)
                throw invalid_argument("linea " + to_string(i + 1) + " con formato incorrecto");
            string turno = "";
            if (c.size() >= 4) turno = c[3];
            Usuario* u = crearUsuario(c[0], c[1], c[2], turno);
            if (buscarUsuario(lista, u->getNombreUsuario()) != -1) {
                delete u;
                throw invalid_argument("usuario repetido en la linea " + to_string(i + 1));
            }
            lista.push_back(u);
        }
    } catch (const exception& e) {
        liberarUsuarios(lista);
        throw runtime_error(string("usuarios.csv: ") + e.what());
    }
    return lista;
}

static void guardarUsuarios(const vector<Usuario*>& lista) {
    vector<string> lineas;
    for (size_t i = 0; i < lista.size(); i++)
        lineas.push_back(lista[i]->serializar()); // polimorfismo
    escribirLineas(rutaUsuarios(), lineas);
}

}

string Habitacion::serializar() const {
    if (!codigoValido(codigo) || capacidad <= 0 ||
        !isfinite(precioPorNoche) || precioPorNoche <= 0)
        throw invalid_argument("No puedes guardar una habitacion incompleta.");
    ostringstream salida;
    salida.precision(17);
    salida << codigo << ',' << nombreTipo(tipo) << ',' << capacidad << ','
           << precioPorNoche << ',' << nombreEstado(estado);
    return salida.str();
}

Habitacion Habitacion::deserializar(const string& linea) {
    stringstream entrada(linea);
    vector<string> campos;
    string campo;
    while (getline(entrada, campo, ',')) campos.push_back(campo);
    if (campos.size() != 5)
        throw invalid_argument("La linea CSV debe tener 5 campos.");

    if (campos[1] != "Sencilla" && campos[1] != "Doble" &&
        campos[1] != "Familiar" && campos[1] != "Suite" &&
        campos[1] != "Otra")
        throw invalid_argument("Tipo desconocido en el archivo.");

    size_t usadosCapacidad = 0, usadosPrecio = 0;
    int capacidad = stoi(campos[2], &usadosCapacidad);
    double precio = stod(campos[3], &usadosPrecio);
    if (usadosCapacidad != campos[2].size() || usadosPrecio != campos[3].size())
        throw invalid_argument("Capacidad o precio con caracteres invalidos.");

    return Habitacion(campos[0], convertirTipoHabitacion(campos[1]),
                      capacidad, precio, convertirEstadoHabitacion(campos[4]));
}

    if (documento.find_first_of(",\r\n") != string::npos ||

        throw invalid_argument("Los datos del huesped no pueden contener comas ni saltos de linea.");

    string d, n, t, c;
    if (!getline(ss, d, ',') || !getline(ss, n, ',') ||
        !getline(ss, t, ',') || !getline(ss, c) ||

    if (!c.empty() && c.back() == '\r') c.pop_back();

    documento = temporal.documento;

// Costo = precio por prenda; si es urgente se suma el recargo por cada prenda.
double Lavanderia::calcularCosto(int cantidad) const {
    if (cantidad <= 0) throw invalid_argument("La cantidad de prendas debe ser mayor a cero.");
    if (!activo) throw invalid_argument("El servicio de lavanderia esta inactivo.");
    double subtotal = precio * cantidad;
    if (urgente) {
        subtotal += recargoUrgente * cantidad;
    }
    return subtotal;
}

string Lavanderia::describir() const {
    string detalle = "[" + codigo + "] " + nombre + " (" + getCategoria() + ") - $" + dinero(precio) + " por prenda";
    if (urgente) detalle += " + $" + dinero(recargoUrgente) + " de recargo por prenda";
    return detalle;
}

    if (tipo != "Lavanderia")

bool textoValido(const string& texto) {
	if (texto.find_first_not_of(" \t") == string::npos) return false; ///< Verifica si el texto es solo espacios o tabulaciones
    if (texto.find_first_of(",\r\n") != string::npos) return false; ///< Verifica si el texto contiene comas, saltos de linea o el \r (retorno de carro)
    return true; 
}

string aMinusculas(string texto) {
    for (size_t i = 0; i < texto.size(); i++)
        texto[i] = (char)tolower((unsigned char)texto[i]); ///< Convierte cada caracter a minuscula usando la funcion tolower()
    return texto;
}

}

bool existeArchivo(const string& ruta) {
    ifstream archivo(ruta.c_str());
    return archivo.is_open();
}

vector<string> leerLineas(const string& ruta) {
    ifstream archivo(ruta.c_str());
    if (!archivo.is_open())
        throw runtime_error("No se pudo abrir el archivo: " + ruta);
    vector<string> lineas;
    string linea;
    while (getline(archivo, linea)) {
        if (!linea.empty() && linea[linea.size() - 1] == '\r')
            linea.erase(linea.size() - 1);
        if (!linea.empty()) lineas.push_back(linea);
    }
    return lineas;
}

void escribirLineas(const string& ruta, const vector<string>& lineas) {
    ofstream archivo(ruta.c_str(), ios::trunc);
    if (!archivo.is_open())
        throw runtime_error("No se pudo escribir el archivo: " + ruta);
    for (size_t i = 0; i < lineas.size(); i++)
        archivo << lineas[i] << "\n";
    archivo.flush();
    if (!archivo)
        throw runtime_error("Error al guardar el archivo: " + ruta);
}

vector<string> dividirCampos(const string& linea) {
    vector<string> campos;
    string campo;
    istringstream ss(linea);
    while (getline(ss, campo, ',')) campos.push_back(campo);
    if (!linea.empty() && linea[linea.size() - 1] == ',') campos.push_back("");
    return campos;
}

int leerOpcion() {
    while (true) {
        cout << "Seleccione una opcion: ";
        string texto;
        if (!getline(cin, texto)) return 0;
        istringstream ss(texto);
        int valor;
        char sobra;
        if ((ss >> valor) && !(ss >> sobra)) return valor;
        cout << "Escribe un numero valido.\n";
    }
}
